⚡ Quick SQL Reference
SELECT
SELECT col1, col2
FROM table
WHERE condition
ORDER BY col DESC
LIMIT 10;
INSERT / UPDATE / DELETE
INSERT INTO t (col) VALUES (val);
UPDATE t SET col=val WHERE id=1;
DELETE FROM t WHERE id=1;
JOINS
SELECT s.name, c.title
FROM students s
INNER JOIN courses c
  ON s.course_id = c.id;
AGGREGATE
SELECT COUNT(*), AVG(marks),
       MAX(marks), MIN(marks)
FROM students
GROUP BY course_id
HAVING AVG(marks) > 60;
